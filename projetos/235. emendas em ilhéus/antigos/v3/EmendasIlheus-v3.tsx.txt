// A bifurcação (peça-chave): tronco com R$ 76 mi e duas estradas, uma por caminho. destaque: 0 nenhum, 1 saúde, 2 Prefeitura, 3 os dois.
const ANG = 0.55;
const Bifurcacao: React.FC<{tema: Tema; pos: V3; escala: number; destaque: number; corS: string; corP: string; ramos: number; rotulos: number}> = ({tema, pos, escala, destaque, corS, corP, ramos, rotulos}) => {
  if (escala < 0.02) return null;
  const luzS = destaque === 1 || destaque === 3 ? 1 : 0;
  const luzP = destaque === 2 || destaque === 3 ? 1 : 0;
  const c = (cor: string, luz: number) => mistura(mistura(tema.objetoEscuro, tema.objeto, 0.4), cor, 0.35 + 0.6 * luz);
  const L = 3.6 * ramos;
  return (
    <group position={pos} scale={S(escala)}>
      <Caixa args={[1.3, 0.06, 3.0]} raio={0.02} cor={mistura(tema.objetoEscuro, tema.objeto, 0.5)} position={[0, 0.03, 1.5]} sombra={false} />
      {ramos > 0.02 ? (
        <>
          <Caixa args={[1.3, 0.07, L]} raio={0.02} cor={c(corS, luzS)} position={[-Math.sin(ANG) * L / 2, 0.035, -Math.cos(ANG) * L / 2]} rotation={[0, ANG, 0]} sombra={false} />
          <Caixa args={[1.3, 0.07, L]} raio={0.02} cor={c(corP, luzP)} position={[Math.sin(ANG) * L / 2, 0.035, -Math.cos(ANG) * L / 2]} rotation={[0, -ANG, 0]} sombra={false} />
        </>
      ) : null}
      <Rotulo3D tema={tema} pos={[-Math.sin(ANG) * 3.9, 0.9, -Math.cos(ANG) * 3.9 - 0.4]} larg={2.6} alt={0.8} texto={['CAMINHO 1', 'SAÚDE']} tam={96} k={rotulos} cor={corS} />
      <Rotulo3D tema={tema} pos={[Math.sin(ANG) * 3.9, 0.9, -Math.cos(ANG) * 3.9 - 0.4]} larg={2.6} alt={0.8} texto={['CAMINHO 2', 'PREFEITURA']} tam={96} k={rotulos} cor={corP} />
    </group>
  );
};

const Barra: React.FC<{tema: Tema; pos: V3; h: number; cor: string; k: number}> = ({tema, pos, h, cor, k}) => {
  const hh = Math.max(0.05, h * k);
  return <Caixa args={[1.0, hh, 1.0]} raio={0.04} cor={mistura(tema.objeto, cor, 0.55)} position={[pos[0], hh / 2, pos[2]]} />;
};
